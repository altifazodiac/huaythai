# 🚨 Emergency Fix 502 Bad Gateway

# 1. หยุด PM2 ทั้งหมด
pm2 stop all
pm2 delete all
pm2 kill

# 2. ล้าง cache
pm2 cleardump
pm2 reset

# 3. Kill processes ที่เกี่ยวข้อง
pkill -f "next"
pkill -f "node.*lottery"
pkill -f "bun.*lottery"

# 4. ตรวจสอบ port
lsof -i:3000
lsof -i:3001
lsof -i:3002

# 5. เริ่มต้นใหม่
pm2 start ecosystem.config.js

# 6. ตรวจสอบสถานะ
pm2 status
pm2 logs --lines 5

# 7. ทดสอบเว็บไซต์
curl -I http://localhost:3000

# หากยังไม่ได้ ให้ลอง:
# - ตรวจสอบ nginx: systemctl status nginx
# - รีสตาร์ท nginx: systemctl restart nginx
# - ตรวจสอบ firewall: ufw status
